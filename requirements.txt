# Runtime + training dependencies, pinned to the versions this project was
# actually developed and tested against on Python 3.12.4.
#
# XGBoost is deliberately absent: it and LightGBM land in the same accuracy
# neighbourhood on tabular data, so running both doubles tuning cost for a
# number that changes no decision. LightGBM wins on this data specifically for
# native categorical support and native NaN routing.

# Core numerics
numpy==1.26.4
pandas==2.2.2
scipy==1.14.0
pyarrow==22.0.0

# Model
# scikit-learn is pinned to 1.3.2, the earliest release with CPython 3.12
# wheels. 1.2.2 (used during development) ships no cp312 wheel and forces a
# source build on every clean install, which is slow and fragile.
lightgbm==4.7.0
xgboost==3.0.5
scikit-learn==1.3.2

# Explainability (powers POST /explain)
# shap 0.52 declares numpy>=2, which conflicts with the numpy 1.26.4 pin above
# and makes the requirement set unresolvable from scratch. 0.46.0 is the newest
# release compatible with numpy 1.x.
shap==0.46.0

# Tuning
optuna==4.9.0

# Experiment tracking (local ./mlruns file store; no server required)
mlflow==3.8.1
